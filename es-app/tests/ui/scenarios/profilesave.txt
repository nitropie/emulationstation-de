# Profile mode: the favourite toggle is captured into the profile state at the metadata save
# point and the shared gamelist keeps its bytes. This fixture saves gamelists on exit, so the
# state write must not depend on that setting.
wait-text "Super Nintendo Entertainment System" 600
assert-text " 3   0"
capture system-view
key return down
key return up
wait-view gamelist 300

key down down
key down up
wait-frames 20
assert-text "Second fixture game."
capture game-selected

key insert down
key insert up
wait-frames 20
assert-text " 3   1"
capture favourite-toggled

quit
