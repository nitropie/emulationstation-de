# First launch: the metadata editor refuses an unencodable personal value and keeps the
# previous one, then a valid value is entered and saved.
wait-text "Super Nintendo Entertainment System" 600
# Normal profile sessions hide shared-library maintenance.
key escape down
key escape up
wait-text "MAIN MENU" 120
assert-text "PROFILE"
assert-absent "SCRAPER"
capture profile-menu
key escape down
key escape up
wait-frames 30
key return down
key return up
wait-view gamelist 300
wait-frames 60
controller back down
controller back up
wait-text "GAMELIST OPTIONS" 300
key down down
key down up
key down down
key down up
key down down
key down up
key return down
key return up
wait-text "TIMES PLAYED" 300
# The editor shows personal fields only and no shared scrape/delete/clear actions.
assert-text "FAVORITE"
assert-text "COMPLETED"
assert-text "HIDDEN"
assert-absent "DESCRIPTION"
assert-absent "SORTNAME"
assert-absent "RELEASE DATE"
assert-absent "DEVELOPER"
assert-absent "PUBLISHER"
assert-absent "GENRE"
assert-absent "RATING"
assert-absent "SCRAPE"
assert-absent "DELETE"
assert-absent "CLEAR"
capture profile-editor
key down down
key down up
key down down
key down up
key down down
key down up
key down down
key down up
wait-frames 20
# Open the field and type something that is not a number.
key return down
key return up
wait-frames 30
capture before-text
text "abc"
wait-frames 10
key return down
key return up
wait-text "IS NOT A VALID VALUE" 300
capture rejected
# The refusal is modal: acknowledge it and the editor keeps the previous value.
controller a down
controller a up
wait-frames 30
# Reopening the field shows the previous value, not the rejected text.
key return down
key return up
wait-frames 30
capture reopened
# Typing appends to the kept value, so the field holds "04", which is the integer 4.
text "4"
wait-frames 10
key return down
key return up
wait-frames 30
capture accepted
# Close the editor and save.
key backspace down
key backspace up
wait-text "SAVE CHANGES?" 300
controller a down
controller a up
wait-frames 60
capture saved
quit
